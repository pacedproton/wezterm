Pattern 1: Creating a Dock Menu (Example from app.rs:131)
──────────────────────────────────────────────────────────
extern "C" fn application_dock_menu(...) -> *mut Object {
    let dock_menu = Menu::new_with_title("");
    let item = MenuItem::new_with(
        "Label",
        Some(sel!(weztermPerformKeyAssignment:)),
        ""
    );
    item.set_represented_item(
        RepresentedItem::KeyAssignment(KeyAssignment::SpawnWindow)
    );
    dock_menu.add_item(&item);
    dock_menu.autorelease()
}

Pattern 2: Dispatching Menu Item Actions (Example from app.rs:96)
──────────────────────────────────────────────────────────────────
extern "C" fn wezterm_perform_key_assignment(
    _self: &mut Object,
    _sel: Sel,
    menu_item: *mut Object,
) {
    let menu_item = MenuItem::with_menu_item(menu_item);
    if let Some(RepresentedItem::KeyAssignment(action)) = 
        menu_item.get_represented_item() {
        if let Some(conn) = Connection::get() {
            conn.dispatch_app_event(
                ApplicationEvent::PerformKeyAssignment(action)
            );
        }
    }
}
